function genCompany(id, isArchive){
    let name=pick(id,NAMES)+pick(id+100,SUFFIX);
    let sector=pick(id+200,SECTORS);
    let hq=pick(id+300,CITIES);
    let arr=srI(id+400,1,50);
    let ret=srI(id+500,40,98);
    let head=srI(id+600,10,500);
    let ltv=srF(id+700,0.5,8.0);
    let winProb = (ret > 80 && ltv > 3.0) ? 0.7 : 0.2;
    if (isArchive) winProb = (ret > 80 && ltv > 3.0) ? 0.1 : 0.01;
    let win = sr(id+800) < winProb;
    return {id, name, sector, hq, arr, ret, head, ltv, win};
}

function getFilteredPortfolio(){
    let q=(document.getElementById('q').value||'').toLowerCase();
    return portfolio.filter(c=>{
        if(q&&!c.name.toLowerCase().includes(q))return false;
        if(activeSectors.length && !activeSectors.includes(c.sector)) return false;
        if(activeCities.length && !activeCities.includes(c.hq)) return false;

        if(activeArrRanges.length){
            let match = activeArrRanges.some(lbl => {
                let r = ARR_RANGES.find(x => x.label === lbl);
                return !(c.arr < r.min || c.arr > r.max);
            });
            if(!match) return false;
        }
        if(activeRetRanges.length){
            let match = activeRetRanges.some(lbl => {
                let r = RET_RANGES.find(x => x.label === lbl);
                return !(c.ret < r.min || c.ret >= r.max);
            });
            if(!match) return false;
        }
        if(activeLtvRanges.length){
            let match = activeLtvRanges.some(lbl => {
                let r = LTV_RANGES.find(x => x.label === lbl);
                return !(c.ltv < r.min || c.ltv >= r.max);
            });
            if(!match) return false;
        }
        return true;
    });
}

    let score=0;
    if(thesis.includes('ARR (Cr)')&&modelWeights['ARR (Cr)'])score+=(c.arr/50)*modelWeights['ARR (Cr)']*50;
    if(thesis.includes('Retention (%)')&&modelWeights['Retention (%)'])score+=(c.ret/100)*modelWeights['Retention (%)']*50;
    if(thesis.includes('Headcount')&&modelWeights['Headcount'])score+=(c.head/500)*modelWeights['Headcount']*50;
    if(thesis.includes('LTV/CAC')&&modelWeights['LTV/CAC'])score+=(c.ltv/8)*modelWeights['LTV/CAC']*50;
    return score;
}
function renderDeals(){
    let sorted=[...deals].sort((a,b)=>scoreDeal(b)-scoreDeal(a));
    let top=sorted.slice(0,5);let rest=sorted.slice(5);
    let renderCard=(c)=>{
        let isSel=selectedDeals.includes(c.id);
        return `<div class="dc ${isSel?'pick':''}" onclick="toggleDeal(${c.id})">
            <div style="font-weight:600;font-family:'DM Sans',sans-serif;margin-bottom:4px">${c.name}</div>
            <div class="note mono" style="margin-bottom:8px">${c.sector} · ${c.hq}</div>
            <div class="mono" style="font-size:11px">ARR ₹${c.arr}Cr · Ret ${c.ret}% · LTV ${c.ltv}x</div>
            <div class="tag" style="background:rgba(27,42,74,.05);color:rgba(27,42,74,.5);margin-top:8px">Model Score: ${scoreDeal(c).toFixed(1)}</div>
        </div>`;
    };
    document.getElementById('shelfA').innerHTML=top.map(renderCard).join('');
    document.getElementById('shelfB').innerHTML=rest.map(renderCard).join('');
    document.getElementById('slots').textContent=selectedDeals.length+'/5';
    document.getElementById('deployBtn').disabled=selectedDeals.length!==5;
}
function toggleDeal(id){
    if(selectedDeals.includes(id))selectedDeals=selectedDeals.filter(x=>x!==id);
    else if(selectedDeals.length<5)selectedDeals.push(id);
    else{toast('Slots Full','You have selected 5 deals.');return;}
    renderDeals();
}

function initPerformance(){
    let chosen=deals.filter(d=>selectedDeals.includes(d.id));
    let returned=0;let hits=0;let rows='';
    chosen.forEach(c=>{
        let ret=c.win?83*(1+srF(c.id+900,1,5)):-83;
        returned+=Math.max(0,ret);if(c.win)hits++;
        rows+=`<tr><td><strong>${c.name}</strong></td><td>${c.sector}</td><td class="r mono">₹83 Cr</td><td class="r mono" style="color:${c.win?'#2D9B5A':'#A32D2D'}">${c.win?'Success':'Write-off'}</td><td class="r mono">₹${Math.max(0,ret).toFixed(1)} Cr</td></tr>`;
    });
    document.getElementById('ret').textContent='₹'+returned.toFixed(1)+' Cr';
    document.getElementById('pnl').textContent='₹'+(returned-415).toFixed(1)+' Cr';
    document.getElementById('hit').textContent=hits+'/5';
    document.getElementById('pf').innerHTML=rows;
    drawPerfChart(chosen);
}
function drawPerfChart(chosen){
    drawGenericCanvas('chartPerf',(ctx,w,h)=>{
        chosen.forEach((c,i)=>{
            let bw=w/chosen.length;let x=i*bw+10;
            let ret=c.win?83*(1+srF(c.id+900,1,5)):0;
            let bh=(ret/400)*(h-40);let y=h-bh-20;
            ctx.fillStyle=c.win?'#2D9B5A':'#A32D2D';roundedRect(ctx,x,y,bw-20,bh,4);ctx.fill();
            ctx.fillStyle='rgba(27,42,74,.35)';ctx.font='10px JetBrains Mono';ctx.textAlign='center';
            ctx.fillText(c.name,x+(bw-20)/2,h-5);
        });
    });
}

function initDebrief(){
    let chosen=deals.filter(d=>selectedDeals.includes(d.id));
    let winners=chosen.filter(c=>c.win);let fails=chosen.filter(c=>!c.win);
    let rows=thesis.map(v=>{
        let key=v.includes('ARR')?'arr':v.includes('Ret')?'ret':v.includes('Head')?'head':'ltv';
        let wMed=winners.length?med(winners.map(c=>c[key])):0;
        let fMed=fails.length?med(fails.map(c=>c[key])):0;
        let lift=wMed-fMed;
        return `<tr><td><strong>${v}</strong></td><td class="r mono">${thesisConf[v]||'-'}</td><td class="r mono">${wMed}</td><td class="r mono">${fMed}</td><td class="r mono" style="color:${lift>0?'#2D9B5A':'#A32D2D'}">${lift>0?'+':''}${lift.toFixed(1)}</td></tr>`;
    }).join('');
    document.getElementById('mirror').innerHTML=rows;
    document.getElementById('fq').textContent=document.getElementById('falsify').value||'No answer provided.';
    document.getElementById('fqn').textContent='In hindsight, the archive showed that high retention alone was not enough without capital efficiency.';
    document.getElementById('truth').innerHTML='<p class="note">The highest performing deals had an LTV/CAC > 3.0 AND Retention > 80%. Companies missing either failed to scale revenue efficiently.</p>';
}
